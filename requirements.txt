# === Core scientific stack ====================================================
numpy                # calcul matriciel & tableaux n-dim ↔ fondation de tout le code
scipy                # fonctions scientifiques (interpol., opti, fft…) utilisées un peu partout
pandas               # manipulations tabulaires (résumés CSV, .h5, etc.)
matplotlib           # génération de toutes les figures / sauvegarde PNG
Pillow               # backend PNG requis par Matplotlib pour fig.savefig(..., format="png")

# === I/O, sérialisation, données =============================================
PyYAML               # lecture/écriture des .yml (catalog_nk.yml, catalog_n2.yml…)
h5py                 # stockage/lecture des fichiers HDF5 « simulation_results.h5 »
joblib               # sauvegarde rapide d’objets Python / parallélisation légère
tqdm                 # barres de progression lors des boucles de simulation
watchdog             # surveillance de fichiers (utils.file_watchers.start_watcher)

# === Jupyter / Widgets / Front-end ============================================
ipykernel            # exécution du kernel Python sous Jupyter / VS Code
IPython              # helpers display, Javascript injection, etc.
ipywidgets           # contrôles interactifs (SimulationTab, MaterialSelector…)
ipyevents            # détection d’événements souris/clavier sur widgets
voila                # déploiement du front-end interactif en dashboard Voila

# === Interface bureau (optionnel) ============================================
PyQt6                # Qt 6 : back-end graphique pour Matplotlib & éventuelles GUI
PyMoosh              # wrapper Qt pour signaux/slots (utilisé par certains widgets avancés)

nbformat             # lecture/écriture programmatique des .ipynb (scripts de patch)

requests             # fetch HTTP dans materials.refractiveindexINFO (télécharge n,k)
